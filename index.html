% 绑定键盘鼠标 set(gcf, 'KeyPressFcn', @key); set(gcf,'WindowButtonDownFcn',@getpoint); % ================== 刷新界面 ================== function wuziqigame(~,~) postion(postion>9)=-9; postion(postion<-9)=9; panding();
